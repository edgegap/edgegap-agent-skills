// Add inside the <Module>(ReadOnlyTargetRules Target) constructor of <Module>.Build.cs,
// after the existing PublicDependencyModuleNames line. The module also needs "HTTP" and "Json"
// (already there if the edgegap-unreal drop-ins are installed).

		// EOS Voice. The server checks player ID tokens with OpenSSL and calls the EOS Voice Web API over
		// HTTP, so it builds for Linux. The client side uses the engine's EOSVoiceChat plugin, which only
		// exists for Windows, Mac and Android clients.
		PrivateDependencyModuleNames.Add("OpenSSL");

		bool bWithEosVoiceClient = Target.Type != TargetType.Server
			&& (Target.Platform == UnrealTargetPlatform.Win64 || Target.Platform == UnrealTargetPlatform.Mac);
		if (bWithEosVoiceClient)
		{
			PrivateDependencyModuleNames.AddRange(new string[] { "EOSSDK", "EOSShared", "VoiceChat", "EOSVoiceChat" });
		}
		PublicDefinitions.Add("WITH_EDGEGAP_EOS_VOICE_CLIENT=" + (bWithEosVoiceClient ? "1" : "0"));
